\begin{lemma}[An isolated crossing forces a zero corner coefficient]
\label{cb:singleton}
Let $A$ be a uniform carrier of $S$, and suppose one of its self-crossing
labels $c$ interlaces no other self-crossing of $A$. Then $c(A)=0$.

\end{lemma}
