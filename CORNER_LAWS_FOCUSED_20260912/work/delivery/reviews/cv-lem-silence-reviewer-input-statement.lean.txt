/-! ## 1. Plane geometry: two parallel closed segments with a common point -/

section Parallel

/-- `det u v = 0` with `u ≠ 0`: `v` is a multiple of `u`. -/
theorem exists_smul_of_det_eq_zero {u v : Plane} (hu : u ≠ 0) (h : det u v = 0) : ∃ c : ℝ, v = c • u := by
  unfold det at h
  by_cases h1 : u.1 = 0
  · have h2 : u.2 ≠ 0 := fun h2 => hu (Prod.ext h1 h2)
    refine ⟨v.2 / u.2, Prod.ext ?_ ?_⟩
    · have hv : u.2 * v.1 = 0 := by rw [h1] at h; linarith
      simp only [Prod.smul_fst, smul_eq_mul, h1, mul_zero]
      exact (mul_eq_zero.mp hv).resolve_left h2
    · simp only [Prod.smul_snd, smul_eq_mul]
      rw [div_mul_cancel₀ _ h2]
  · refine ⟨v.1 / u.1, Prod.ext ?_ ?_⟩
    · simp only [Prod.smul_fst, smul_eq_mul]
      rw [div_mul_cancel₀ _ h1]
    · simp only [Prod.smul_snd, smul_eq_mul]
      field_simp
      linarith

/-- **Two parallel closed edge segments with a common point have an endpoint of one on the other**
(the geometric content of rem:silentclauses, d1_setup.tex:1275–1292: "since they are parallel … they meet at a
point that is an endpoint of at least one of them"). -/
theorem parallel_segments_meet_endpoint {P : LabelledTuple n} {e f : ZMod n}
    (he : edge P e ≠ 0) (hf : edge P f ≠ 0) (hdet : det (edge P e) (edge P f) = 0)
    {x : Plane} (hxe : x ∈ edgeSegment P e) (hxf : x ∈ edgeSegment P f) :
    P f ∈ edgeSegment P e ∨ P (f + 1) ∈ edgeSegment P e ∨
      P e ∈ edgeSegment P f ∨ P (e + 1) ∈ edgeSegment P f := by
  obtain ⟨c, hc⟩ := exists_smul_of_det_eq_zero he hdet
  have hc0 : c ≠ 0 := by
    rintro rfl
    rw [zero_smul] at hc
    exact hf hc
  obtain ⟨s, hs0, hs1, hsx⟩ := hxe
  obtain ⟨t, ht0, ht1, htx⟩ := hxf
  -- the four endpoints, each in the other segment's parametrization
  have hPf : P f = edgePoint P e (s - t * c) := by
    have h1 : P e + s • edge P e = P f + (t * c) • edge P e := by
      rw [← smul_smul, ← hc]
      exact hsx.symm.trans htx
    unfold edgePoint
    rw [sub_smul, ← add_sub_assoc, h1]
    abel
  have hPf1 : P (f + 1) = edgePoint P e (s - t * c + c) := by
    have h1 : P (f + 1) = P f + edge P f := by simp [edge]
    rw [h1, hPf, hc]
    unfold edgePoint
    rw [add_smul, add_assoc]
  have hPe : P e = edgePoint P f (-(s - t * c) / c) := by
    unfold edgePoint
    rw [hc, smul_smul, div_mul_cancel₀ _ hc0, hPf]
    unfold edgePoint
    rw [neg_smul, add_neg_cancel_right]
  have hPe1 : P (e + 1) = edgePoint P f ((1 - (s - t * c)) / c) := by
    have h1 : P (e + 1) = P e + edge P e := by simp [edge]
    unfold edgePoint
    rw [hc, smul_smul, div_mul_cancel₀ _ hc0, hPf, h1]
    unfold edgePoint
    rw [sub_smul (1 : ℝ), one_smul]
    abel
  set β := s - t * c with hβ
  have hs : s = (1 - t) * β + t * (β + c) := by rw [hβ]; ring
  by_cases hβin : 0 ≤ β ∧ β ≤ 1
  · exact Or.inl ⟨β, hβin.1, hβin.2, hPf⟩
  by_cases hβcin : 0 ≤ β + c ∧ β + c ≤ 1
  · exact Or.inr (Or.inl ⟨β + c, hβcin.1, hβcin.2, hPf1⟩)
  -- both endpoints of `f` are off `[0,1]`: since `s ∈ [0,1]` is a convex combination of them, they lie on
  -- opposite sides, and then `0` (the point `P e`) lies between them
  have hkey : (β < 0 ∧ 1 < β + c) ∨ (1 < β ∧ β + c < 0) := by
    rcases lt_or_ge β 0 with hβneg | hβnn
    · rcases lt_or_ge 1 (β + c) with hbig | hsmall
      · exact Or.inl ⟨hβneg, hbig⟩
      · exfalso
        have hβc : β + c < 0 := by
          rcases lt_or_ge (β + c) 0 with h | h
          · exact h
          · exact absurd ⟨h, hsmall⟩ hβcin
        have h2 : t * (β + c) ≤ 0 := mul_nonpos_of_nonneg_of_nonpos ht0 hβc.le
        rcases lt_or_eq_of_le ht1 with htlt | hteq
        · have h1 : (1 - t) * β < 0 := mul_neg_of_pos_of_neg (by linarith) hβneg
          linarith
        · rw [hteq] at hs
          linarith
    · have hβbig : 1 < β := by
        rcases lt_or_ge 1 β with h | h
        · exact h
        · exact absurd ⟨hβnn, h⟩ hβin
      rcases lt_or_ge (β + c) 0 with hneg | hnn
      · exact Or.inr ⟨hβbig, hneg⟩
      · exfalso
        have hβc : 1 < β + c := by
          rcases lt_or_ge 1 (β + c) with h | h
          · exact h
          · exact absurd ⟨hnn, h⟩ hβcin
        have h2 : t * 1 ≤ t * (β + c) := mul_le_mul_of_nonneg_left hβc.le ht0
        rcases lt_or_eq_of_le ht1 with htlt | hteq
        · have h1 : (1 - t) * 1 < (1 - t) * β := mul_lt_mul_of_pos_left hβbig (by linarith)
          linarith
        · rw [hteq] at hs
          linarith
  rcases hkey with ⟨hβneg, hbig⟩ | ⟨hβbig, hneg⟩
  · have hcpos : 0 < c := by linarith
    refine Or.inr (Or.inr (Or.inl ⟨-β / c, div_nonneg (by linarith) hcpos.le, ?_, hPe⟩))
    rw [div_le_one hcpos]
    linarith
  · have hcneg : c < 0 := by linarith
    refine Or.inr (Or.inr (Or.inl ⟨-β / c, div_nonneg_of_nonpos (by linarith) hcneg.le, ?_, hPe⟩))
    rw [div_le_one_of_neg hcneg]
    linarith

/-! ## 2. Activation is an open condition -/

/-- The activation relation `Crosses` (CV:def:guarded, the strict four-sign test) is open in the polygon. -/
theorem Crosses.eventually {P : LabelledTuple n} {e f : ZMod n} (h : Crosses P e f) :
    ∀ᶠ Q in 𝓝 P, Crosses Q e f := by
  obtain ⟨hr, h1, h2⟩ := h
  have c1 : Continuous fun Q : LabelledTuple n => G2 Q e f * G2 Q e (f + 1) :=
    (continuous_G2 e f).mul (continuous_G2 e (f + 1))
  have c2 : Continuous fun Q : LabelledTuple n => G2 Q f e * G2 Q f (e + 1) :=
    (continuous_G2 f e).mul (continuous_G2 f (e + 1))
  filter_upwards [c1.continuousAt.eventually (isOpen_Iio.mem_nhds h1),
    c2.continuousAt.eventually (isOpen_Iio.mem_nhds h2)] with Q hQ1 hQ2
  exact ⟨hr, hQ1, hQ2⟩

end Parallel

/-- Activation of a conditional member (CV:def:guarded) is open in the polygon. -/
theorem Member.Active.eventually {P : LabelledTuple n} {m : Member n} (h : m.Active P) :
    ∀ᶠ Q in 𝓝 P, m.Active Q := by
  cases m with
  | g1 i => simp only [Member.Active] at h
  | g2 e i _ => simp only [Member.Active] at h
  | g5 e f _ => exact Crosses.eventually h
  | g3 e f g _ =>
    obtain ⟨h1, h2, h3⟩ := h
    filter_upwards [Crosses.eventually h1, Crosses.eventually h2, Crosses.eventually h3] with Q hQ1 hQ2 hQ3
    exact ⟨hQ1, hQ2, hQ3⟩
  | g4 e f g _ =>
    obtain ⟨h1, h2⟩ := h
    filter_upwards [Crosses.eventually h1, Crosses.eventually h2] with Q hQ1 hQ2
    exact ⟨hQ1, hQ2⟩

theorem exists_punctured : ∃ t : E.Parameter, t.val ≠ 0 :=

/-- **A member active at the centre is relevant at some `t ≠ 0`** (activation is open and the curve is
continuous): the step "an active member is relevant, hence lies in `Z` if it vanishes at `t = 0`"

theorem exists_punctured_relevant_of_active (m : Member n) (h : m.Active E.center) :
    ∃ t : E.Parameter, t.val ≠ 0 ∧ m.Relevant (E.curve t) := by
  have hev : ∀ᶠ t in 𝓝 E.zeroParameter, m.Active (E.curve t) :=
    E.continuous_curve.continuousAt.eventually h.eventually
  obtain ⟨δ, hδ, hδr, hall⟩ := (E.eventually_center_iff_radius _).mp hev
  refine ⟨⟨δ / 2, by constructor <;> linarith⟩, (half_pos hδ).ne', Or.inr (hall _ ?_)⟩
  show |δ / 2| < δ
  rw [abs_of_pos (half_pos hδ)]
  exact half_lt_self hδ

/-- "The turn sign at a vertex is `sgn G1_i`, which is unconditional, hence relevant, and lies in no `Z` of a

theorem g1_ne_zero (i : ZMod n) : G1 E.center i ≠ 0 := by
  obtain ⟨t, ht⟩ := E.exists_punctured
  have h := (guardconst E (Member.g1 i) ⟨t, ht, Member.relevant_g1 _ i⟩
    (fun hZ => hE.no_g1 _ hZ i rfl)).1
  simpa using h

/-- The turns of the centre are nonzero. -/
theorem turn_ne_zero (i : ZMod n) : turn E.center i ≠ 0 := by

  exact sign_ne_zero.mpr (hE.g1_ne_zero i)

/-- **No vertex of the centre lies on a non-incident closed edge**: "a member of `Z` may put `p_i` on the line of
`e` but never on the segment — and every other relevant `G2` is nonzero throughout by Lemma lem:guardconst"
(d1_setup.tex:1316–1318). -/
theorem vertex_not_mem_edge {k e : ZMod n} (hk : ¬ incident k e) :
    E.center k ∉ edgeSegment E.center e := by
  obtain ⟨h0, h1⟩ := (nonincident_iff k e).mp hk
  intro hmem
  by_cases hZ : Member.g2 e k ⟨h0, h1⟩ ∈ E.zeroSet
  · obtain ⟨e', i', h', heq, -, hseg⟩ := hE _ hZ
    cases heq
    exact hseg hmem
  · obtain ⟨t, ht⟩ := E.exists_punctured
    have h := (guardconst E _ ⟨t, ht, Member.relevant_g2 _ e k ⟨h0, h1⟩⟩ hZ).1
    apply h
    rw [Member.eval_g2]
    exact lineForm_eq_zero_of_mem E.center e hmem

theorem vertex_not_mem_edge' : ∀ k e : ZMod n, ¬ incident k e → E.center k ∉ edgeSegment E.center e :=
  fun _ _ hk => hE.vertex_not_mem_edge hk

/-- **Remote segments of the centre meeting at a point are transverse**: parallel closed segments with a
common point would put an endpoint of one on the other (`parallel_segments_meet_endpoint`), which the previous
clause excludes (rem:silentclauses, d1_setup.tex:1283–1292). -/
theorem remote_transverse {e f : ZMod n} (hr : remote e f) {x : Plane}
    (hxe : x ∈ edgeSegment E.center e) (hxf : x ∈ edgeSegment E.center f) :
    det (edge E.center e) (edge E.center f) ≠ 0 := by
  intro hdet
  obtain ⟨h0, h1, h2, h3⟩ := remote_endpoints e f hr
  rcases parallel_segments_meet_endpoint (E.center_isPolygon e) (E.center_isPolygon f) hdet hxe hxf
    with h | h | h | h
  · exact hE.vertex_not_mem_edge ((nonincident_iff f e).mpr ⟨h0, h1⟩) h
  · exact hE.vertex_not_mem_edge ((nonincident_iff (f + 1) e).mpr ⟨h2, h3⟩) h
  · exact hE.vertex_not_mem_edge ((nonincident_iff e f).mpr ⟨h0.symm, h2.symm⟩) h
  · exact hE.vertex_not_mem_edge ((nonincident_iff (e + 1) f).mpr ⟨h1.symm, h3.symm⟩) h

/-- Remote segments of the centre that meet are an active pair (`Crosses`, the strict four-sign test): the
meeting point is interior to both (`remote_closed_point_interior`) and transverse. -/
theorem crosses_of_segments_meet {e f : ZMod n} (hr : remote e f) {x : Plane}
    (hxe : x ∈ edgeSegment E.center e) (hxf : x ∈ edgeSegment E.center f) : Crosses E.center e f := by
  rw [crosses_iff]
  obtain ⟨s, hs0, hs1, hsx⟩ := remote_closed_point_interior hE.vertex_not_mem_edge' hr hxe hxf
  obtain ⟨t, ht0, ht1, htx⟩ :=
    remote_closed_point_interior hE.vertex_not_mem_edge' (remote_symm hr) hxf hxe
  exact ⟨hr, s, t, hs0, hs1, ht0, ht1, hsx.symm.trans htx, hE.remote_transverse hr hxe hxf⟩

/-- **No sorted triple of pairwise remote edges of the centre is concurrent** (d1_setup.tex:1320–1326): the
three pairwise meetings are active, so the (G3) member is active at the centre, hence relevant at some `t ≠ 0`;
it vanishes at the centre (`G3_eq_zero_of_common_point`), so it would be a member of `Z`, "which Definition
def:silent forbids, its zero sets containing no (G3)". -/
theorem no_sorted_triple {e f g : ZMod n}
    (h : remote e f ∧ remote f g ∧ remote e g ∧ rep e < rep f ∧ rep f < rep g)
    {x : Plane} (he : x ∈ edgeSegment E.center e) (hf : x ∈ edgeSegment E.center f)
    (hg : x ∈ edgeSegment E.center g) : False := by
  have hact : (Member.g3 e f g h).Active E.center := by
    simp only [Member.Active]
    exact ⟨hE.crosses_of_segments_meet h.1 he hf, hE.crosses_of_segments_meet h.2.1 hf hg,
      hE.crosses_of_segments_meet h.2.2.1 he hg⟩
  obtain ⟨t, ht, hrel⟩ := E.exists_punctured_relevant_of_active _ hact
  have hZ : Member.g3 e f g h ∈ E.zeroSet := by
    refine ⟨?_, t, ht, hrel⟩
    rw [Member.eval_g3]
    exact G3_eq_zero_of_common_point E.center e f g (lineForm_eq_zero_of_mem _ _ he)
      (lineForm_eq_zero_of_mem _ _ hf) (lineForm_eq_zero_of_mem _ _ hg)
  exact hE.no_g3 _ hZ e f g h rfl

/-- No three pairwise remote closed edge segments of the centre share a point (the sorted case, in each of the
six orders of the representatives). -/
theorem no_remote_triple {i j k : ZMod n} (hij : remote i j) (hjk : remote j k) (hik : remote i k) :
    ¬ ClosedTripleMeet E.center i j k := by
  rintro ⟨x, hi, hj, hk⟩
  have hij' : i ≠ j := (remote_endpoints i j hij).1.symm
  have hjk' : j ≠ k := (remote_endpoints j k hjk).1.symm
  have hik' : i ≠ k := (remote_endpoints i k hik).1.symm
  rcases rep_lt_or_lt hij' with h1 | h1 <;> rcases rep_lt_or_lt hjk' with h2 | h2 <;>
    rcases rep_lt_or_lt hik' with h3 | h3
  · exact hE.no_sorted_triple ⟨hij, hjk, hik, h1, h2⟩ hi hj hk
  · exact absurd (h1.trans h2) (lt_asymm h3)
  · exact hE.no_sorted_triple ⟨hik, remote_symm hjk, hij, h3, h2⟩ hi hk hj
  · exact hE.no_sorted_triple ⟨remote_symm hik, hij, remote_symm hjk, h3, h1⟩ hk hi hj
  · exact hE.no_sorted_triple ⟨remote_symm hij, hik, hjk, h1, h3⟩ hj hi hk
  · exact hE.no_sorted_triple ⟨hjk, remote_symm hik, remote_symm hij, h2, h3⟩ hj hk hi
  · exact absurd (h2.trans h1) (lt_asymm h3)
  · exact hE.no_sorted_triple ⟨remote_symm hjk, remote_symm hij, remote_symm hik, h2, h1⟩ hk hj hi

/-- SM's (G2) at the centre: no three edge interiors concur (`weak_base_g2_iff_no_remote_closed_triples`). -/
theorem g2_sm : SM.G2 E.center :=
  (weak_base_g2_iff_no_remote_closed_triples E.center_isPolygon hE.turn_ne_zero
    hE.vertex_not_mem_edge').mpr fun _ _ _ hij hjk hik => hE.no_remote_triple hij hjk hik

/-- **The centre of a silent event is weakly generic** (`SM.WeakGeneric`, the accepted tier 2): the CV analogue
of the accepted `SM.WallGerm.silent_center_weak` (SM/SilentCenter.lean:40–46). Edges nonzero
(`center_polygon`), turns nonzero (G1 off `Z`), vertices off non-incident segments (G2 off `Z`, or in `Z` with the
vertex off the segment by def:silent), remote meetings transverse and unique, no three interiors concurrent. -/
theorem silent_center_weakGeneric (E : Event n) (hE : E.Silent) : WeakGeneric E.center :=
  ⟨E.center_isPolygon, hE.turn_ne_zero, hE.vertex_not_mem_edge',
    fun _ _ hr => ⟨fun _ hi hj => hE.remote_transverse hr hi hj,
      fun _ _ hxi hxj hyi hyj =>
        transverse_segments_unique (hE.remote_transverse hr hxi hxj) hxi hxj hyi hyj⟩,
    hE.g2_sm⟩

/-- The centre of a silent event has crossing geometry (tier 0). -/
theorem silent_center_crossingGeometry (E : Event n) (hE : E.Silent) : CrossingGeometry E.center :=
  weak_crossingGeometry (silent_center_weakGeneric E hE)

/-- The centre of a silent event is diagrammatic (CV:def:diagrammatic) for `n ≥ 3`. -/
theorem silent_center_diagrammatic (hn : 3 ≤ n) (E : Event n) (hE : E.Silent) : Diagrammatic E.center :=
  diagrammatic_of_weak hn (silent_center_weakGeneric E hE)

/-- **Every polygon of a silent event is weakly generic**: generic off the centre (def:event), weakly generic
at it. The whole interval `(−ε, ε)` lies in the weak locus — no radius is shrunk (the decision's
`silent_curve_weakGeneric` with `δ = ε`). -/
theorem silent_curve_weakGeneric (E : Event n) (hE : E.Silent) (t : E.Parameter) :
    WeakGeneric (E.curve t) := by
  by_cases ht : t.val = 0
  · have h0 : t = E.zeroParameter := Subtype.ext ht
    rw [h0]
    exact silent_center_weakGeneric E hE
  · exact (E.generic_punctured t ht).weakGeneric

theorem paramSegment_mem (tm tp : E.Parameter) (u : unitInterval) :
    tm.val + u.val * (tp.val - tm.val) ∈ Set.Ioo (-E.radius) E.radius := by
  obtain ⟨hm0, hm1⟩ := tm.property
  obtain ⟨hp0, hp1⟩ := tp.property
  have hu0 := unitInterval.nonneg u
  have hu1 := unitInterval.le_one u

    have h2 := mul_nonneg (sub_nonneg.mpr hu1) (sub_nonneg.mpr h)

  · have h1 := mul_nonpos_of_nonneg_of_nonpos hu0 (sub_nonpos.mpr h)
    have h2 := mul_nonpos_of_nonneg_of_nonpos (sub_nonneg.mpr hu1) (sub_nonpos.mpr h)
    constructor <;> nlinarith

theorem silence_near (hn : 3 ≤ n) (E : Event n) (hE : E.Silent) :
