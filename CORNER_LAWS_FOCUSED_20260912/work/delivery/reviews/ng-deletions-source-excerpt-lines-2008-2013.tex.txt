\begin{lemma}[Zigzag and crossed-cusp deletion]
\label{ng:deletions}
Deleting an empty zigzag lowers $s$ by two and cannot increase $B$; applying
the crossed-cusp shortcut lowers $s$ by one and cannot increase $B$.

\end{lemma}
